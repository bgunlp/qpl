  AND chartevents.itemid IN

          DENSE_RANK() OVER (
                             ORDER BY COUNT_BIG(*) DESC) AS C1
   FROM
     (SELECT admissions.subject_id AS subject_id,
             diagnoses_icd.charttime AS charttime
      FROM diagnoses_icd
      JOIN admissions ON diagnoses_icd.hadm_id = admissions.hadm_id
      WHERE diagnoses_icd.icd_code =
          (SELECT d_icd_diagnoses.icd_code
           FROM d_icd_diagnoses
           WHERE d_icd_diagnoses.long_title = 'dependence on respirator, status')
        AND dbo.MET_BY(diagnoses_icd.charttime, dbo.TIME_POINT_START_OF_YMD(2100, NULL, NULL)) = 1) AS T1
   JOIN
     (SELECT admissions.subject_id AS subject_id,
             microbiologyevents.test_name AS test_name,

   WHERE T1.charttime < T2.charttime
     AND dbo.DURING(T1.charttime, dbo.TIME_INTERVAL_REL(T2.charttime, 'month')) = 1
